\section{Вывести канонические решения для}
Дифференциальное уравнение

\begin{equation*}
    a_{11} \frac{\partial^2 u}{\partial x^2} + 2a_{12} \frac{\partial^2 u}{\partial x \partial y} + a_{22} \frac{\partial^2 u}{\partial y^2} + F\left(x,y,u,\frac{\partial u}{\partial x},\frac{\partial u}{\partial y}\right) = 0
\end{equation*}

Характеристическое уравнение

\begin{equation*}
    \frac{dy}{dx} = \frac{a_{12} \pm \sqrt{a_{12}^2 - a_{11}a_{22}}}{a_{11}}
\end{equation*}

\begin{itemize}
    \item Гиперболических уравнений$ \quad a^2_{12} - a_{11}a_{22} > 0$
    \item Параболических уравнений$ \quad a^2_{12} - a_{11}a_{22} = 0$
    \item Эллиптических уравнений$ \quad a^2_{12} - a_{11}a_{22} < 0$
\end{itemize}
